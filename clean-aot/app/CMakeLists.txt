cmake_minimum_required(VERSION 3.20)
project(clean_aot_app LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(CLEAN_AOT_APP_BUILD_TESTS "Build application-shell tests" ON)
option(CLEAN_AOT_APP_ENABLE_SDL "Build the native SDL platform adapter" OFF)
option(CLEAN_AOT_APP_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
set(CLEAN_AOT_PRODUCT_GENERATED_DIR "" CACHE PATH "External locally generated clean BIOS region sources")
set(CLEAN_AOT_PRODUCT_GENERATED_DIRS "" CACHE STRING "Additional setup-generated authenticated epoch source directories")
set(CLEAN_AOT_PRODUCT_BINDING_SYMBOL "" CACHE STRING "Generated BIOS binding accessor identifier")
set(CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS "" CACHE STRING "Setup-generated authenticated epoch binding accessors")
set(CLEAN_AOT_PRODUCT_AUTHENTICATED_EVIDENCE_DIR "" CACHE PATH "Committed authenticated frontier receipt directory")
set(CLEAN_AOT_PRODUCT_GENERATED_ROOT "" CACHE PATH "Setup-generated unique region artifact root")
set(CLEAN_AOT_PRODUCT_GAME_SHARD_EVIDENCE "" CACHE FILEPATH "Authenticated game shard receipt")

include(CTest)
set(_clean_aot_app_build_testing "${BUILD_TESTING}")
set(BUILD_TESTING OFF)
set(CLEAN_AOT_HOST_BUILD_TESTS OFF CACHE BOOL "" FORCE)

if(NOT TARGET clean_aot_runtime)
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../runtime"
                     "${CMAKE_CURRENT_BINARY_DIR}/runtime")
endif()
set(BUILD_TESTING "${_clean_aot_app_build_testing}")
if(NOT TARGET clean_aot_host)
    if(CLEAN_AOT_APP_ENABLE_SDL)
        set(CLEAN_AOT_HOST_ENABLE_SDL_VULKAN ON CACHE BOOL "" FORCE)
    endif()
    add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../host"
                     "${CMAKE_CURRENT_BINARY_DIR}/host")

endif()
target_compile_features(clean_aot_app PUBLIC cxx_std_20)

function(clean_aot_app_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(${target} PRIVATE
            -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Wshadow -Werror)
    endif()
    if(CLEAN_AOT_APP_ENABLE_SANITIZERS AND NOT MSVC)
        target_compile_options(${target} PRIVATE
            -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
        target_link_options(${target} PRIVATE
            -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
endfunction()
clean_aot_app_warnings(clean_aot_app)

if(CLEAN_AOT_APP_ENABLE_SDL)
    add_library(clean_aot_app_sdl STATIC src/sdl_platform.cpp)
    target_include_directories(clean_aot_app_sdl PUBLIC include)
    target_link_libraries(clean_aot_app_sdl PUBLIC clean_aot_app
        clean_aot_host_sdl_vulkan SDL3::SDL3)
    clean_aot_app_warnings(clean_aot_app_sdl)
    add_executable(Wipeout3SE src/product_main.cpp)
    if(MINGW)
        target_link_options(Wipeout3SE PRIVATE -municode)
    endif()
    target_link_libraries(Wipeout3SE PRIVATE clean_aot_app clean_aot_runtime
        clean_aot_host_sdl_vulkan SDL3::SDL3)
    if(WIN32)
        target_link_libraries(Wipeout3SE PRIVATE comdlg32)
    endif()
    if(CLEAN_AOT_PRODUCT_GENERATED_DIR)
        file(GLOB _clean_aot_product_generated CONFIGURE_DEPENDS
            "${CLEAN_AOT_PRODUCT_GENERATED_DIR}/*.cpp")
        if(NOT _clean_aot_product_generated)
            message(FATAL_ERROR "CLEAN_AOT_PRODUCT_GENERATED_DIR contains no generated sources")
        endif()
        target_sources(Wipeout3SE PRIVATE ${_clean_aot_product_generated})
    endif()
    if(CLEAN_AOT_PRODUCT_AUTHENTICATED_EVIDENCE_DIR OR CLEAN_AOT_PRODUCT_GENERATED_ROOT)
        if(NOT CLEAN_AOT_PRODUCT_AUTHENTICATED_EVIDENCE_DIR OR NOT CLEAN_AOT_PRODUCT_GENERATED_ROOT)
            message(FATAL_ERROR "Authenticated evidence and generated root must be configured together")
        endif()
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        set(_clean_aot_registry_dir "${CMAKE_CURRENT_BINARY_DIR}/epoch-registry")
        set(_clean_aot_registry_command "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/product/generate_epoch_registry.py"
            --evidence-dir "${CLEAN_AOT_PRODUCT_AUTHENTICATED_EVIDENCE_DIR}"
            --generated-root "${CLEAN_AOT_PRODUCT_GENERATED_ROOT}"
            --output "${_clean_aot_registry_dir}")
        if(CLEAN_AOT_PRODUCT_GAME_SHARD_EVIDENCE)
            list(APPEND _clean_aot_registry_command --game-shards
                "${CLEAN_AOT_PRODUCT_GAME_SHARD_EVIDENCE}")
        endif()
        execute_process(COMMAND ${_clean_aot_registry_command}
            RESULT_VARIABLE _clean_aot_registry_result ERROR_VARIABLE _clean_aot_registry_error)
        if(NOT _clean_aot_registry_result EQUAL 0)
            message(FATAL_ERROR "Authenticated epoch registry generation failed: ${_clean_aot_registry_error}")
        endif()
        include("${_clean_aot_registry_dir}/epoch-registry.cmake")
        target_sources(Wipeout3SE PRIVATE ${CLEAN_AOT_REGISTRY_SOURCES})
        target_include_directories(Wipeout3SE PRIVATE "${_clean_aot_registry_dir}")
        target_compile_definitions(Wipeout3SE PRIVATE CLEAN_AOT_PRODUCT_HAS_EPOCH_BINDINGS=1)
    endif()
    foreach(_clean_aot_epoch_dir IN LISTS CLEAN_AOT_PRODUCT_GENERATED_DIRS)
        file(GLOB _clean_aot_epoch_generated CONFIGURE_DEPENDS "${_clean_aot_epoch_dir}/*.cpp")
        if(NOT _clean_aot_epoch_generated)
            message(FATAL_ERROR "Authenticated epoch directory contains no generated sources: ${_clean_aot_epoch_dir}")
        endif()
        target_sources(Wipeout3SE PRIVATE ${_clean_aot_epoch_generated})
    endforeach()
    list(LENGTH CLEAN_AOT_PRODUCT_GENERATED_DIRS _clean_aot_epoch_dir_count)
    list(LENGTH CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS _clean_aot_epoch_symbol_count)
    if(NOT _clean_aot_epoch_dir_count EQUAL _clean_aot_epoch_symbol_count)
        message(FATAL_ERROR "Authenticated epoch directories and binding accessors must match one-for-one")
    endif()
    if(CLEAN_AOT_PRODUCT_BINDING_SYMBOL)
        target_compile_definitions(Wipeout3SE PRIVATE
            CLEAN_AOT_PRODUCT_BINDING_SYMBOL=${CLEAN_AOT_PRODUCT_BINDING_SYMBOL})
    endif()
    if(CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS AND NOT CLEAN_AOT_PRODUCT_GENERATED_ROOT)
        set(_clean_aot_epoch_header "${CMAKE_CURRENT_BINARY_DIR}/product_epoch_bindings.hpp")
        file(WRITE "${_clean_aot_epoch_header}" "#pragma once\n#include <array>\n#include <clean_aot/runtime/runtime.hpp>\n")
        foreach(_clean_aot_symbol IN LISTS CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS)
            file(APPEND "${_clean_aot_epoch_header}" "extern \"C\" const clean_aot::runtime::GeneratedExecutableBinding* ${_clean_aot_symbol}() noexcept;\n")
        endforeach()
        list(LENGTH CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS _clean_aot_epoch_count)
        file(APPEND "${_clean_aot_epoch_header}" "inline std::array<const clean_aot::runtime::GeneratedExecutableBinding*, ${_clean_aot_epoch_count}> clean_aot_product_epoch_bindings() noexcept { return {")
        foreach(_clean_aot_symbol IN LISTS CLEAN_AOT_PRODUCT_EPOCH_BINDING_SYMBOLS)
            file(APPEND "${_clean_aot_epoch_header}" "${_clean_aot_symbol}(),")
        endforeach()
        file(APPEND "${_clean_aot_epoch_header}" "}; }\n")
        file(APPEND "${_clean_aot_epoch_header}" "inline std::array<const clean_aot::runtime::GeneratedExecutableBinding*, 0> clean_aot_product_game_bindings() noexcept { return {}; }\n")
        target_include_directories(Wipeout3SE PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
        target_compile_definitions(Wipeout3SE PRIVATE CLEAN_AOT_PRODUCT_HAS_EPOCH_BINDINGS=1)
    endif()
    if(MSVC)
        target_link_options(Wipeout3SE PRIVATE "/MAP:$<TARGET_FILE_DIR:Wipeout3SE>/Wipeout3SE.map")
    else()
        target_link_options(Wipeout3SE PRIVATE "-Wl,-Map,$<TARGET_FILE_DIR:Wipeout3SE>/Wipeout3SE.map")
    endif()
    clean_aot_app_warnings(Wipeout3SE)
    clean_aot_colocate_sdl3(Wipeout3SE)
    if(CLEAN_AOT_APP_BUILD_TESTS AND BUILD_TESTING)
        add_executable(clean_aot_app_sdl_tests tests/sdl_platform_tests.cpp)
        target_link_libraries(clean_aot_app_sdl_tests PRIVATE clean_aot_app_sdl)
        clean_aot_app_warnings(clean_aot_app_sdl_tests)
        clean_aot_colocate_sdl3(clean_aot_app_sdl_tests)
        if(WIN32)
            add_executable(clean_aot_windows_test_launcher
                tests/windows_test_launcher.cpp)
            clean_aot_app_warnings(clean_aot_windows_test_launcher)
            add_test(NAME clean_aot_app_sdl_tests
                COMMAND clean_aot_windows_test_launcher
                    $<TARGET_FILE:clean_aot_app_sdl_tests>
                    $<TARGET_FILE_DIR:SDL3::SDL3>)
        else()
            add_test(NAME clean_aot_app_sdl_tests COMMAND clean_aot_app_sdl_tests)
        endif()
        set_tests_properties(clean_aot_app_sdl_tests PROPERTIES
            ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
    endif()
endif()
